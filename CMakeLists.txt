cmake_minimum_required(VERSION 3.28)
project(ParticleWorld VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS")
endif()
option(PW_BUILD_APP "Build the native GPU application" ON)
set(PW_WORKGROUP_SIZE "128" CACHE STRING "Bulk compute threads per workgroup (128 or 256)")
set_property(CACHE PW_WORKGROUP_SIZE PROPERTY STRINGS 128 256)
if(NOT PW_WORKGROUP_SIZE MATCHES "^(128|256)$")
  message(FATAL_ERROR "PW_WORKGROUP_SIZE must be 128 or 256")
endif()
include(CTest)
add_library(pw_core INTERFACE)
target_include_directories(pw_core INTERFACE src)
target_compile_features(pw_core INTERFACE cxx_std_20)
target_compile_definitions(pw_core INTERFACE PW_WORKGROUP_SIZE=${PW_WORKGROUP_SIZE})
if(BUILD_TESTING)
  add_executable(pw_core_tests tests/unit/core_tests.cpp)
  target_link_libraries(pw_core_tests PRIVATE pw_core)
  add_test(NAME core_invariants COMMAND pw_core_tests)
  add_executable(pw_voxel_coordinates_tests tests/unit/voxel_coordinates_tests.cpp)
  target_link_libraries(pw_voxel_coordinates_tests PRIVATE pw_core)
  add_test(NAME voxel_coordinates COMMAND pw_voxel_coordinates_tests)

  target_link_libraries(pw_sparse_voxel_store_tests PRIVATE pw_core)
  add_test(NAME sparse_voxel_store COMMAND pw_sparse_voxel_store_tests)
  add_executable(pw_spatial_grid_tests tests/unit/spatial_grid_tests.cpp)
  target_link_libraries(pw_spatial_grid_tests PRIVATE pw_core)
  add_test(NAME spatial_grid_reference COMMAND pw_spatial_grid_tests)
  add_executable(pw_spatial_index_tests tests/unit/spatial_index_tests.cpp)
  target_link_libraries(pw_spatial_index_tests PRIVATE pw_core)
  add_test(NAME spatial_index_reference COMMAND pw_spatial_index_tests)
  add_executable(pw_neighbor_reference_tests tests/unit/neighbor_reference_tests.cpp)
  target_link_libraries(pw_neighbor_reference_tests PRIVATE pw_core)
  add_test(NAME neighbor_reference COMMAND pw_neighbor_reference_tests)
  add_executable(pw_contact_normal_tests tests/unit/contact_normal_tests.cpp)
  target_link_libraries(pw_contact_normal_tests PRIVATE pw_core)
  add_test(NAME contact_normal_reference COMMAND pw_contact_normal_tests)
  add_executable(pw_contact_velocity_tests tests/unit/contact_velocity_tests.cpp)
  target_link_libraries(pw_contact_velocity_tests PRIVATE pw_core)
  add_test(NAME contact_velocity_reference COMMAND pw_contact_velocity_tests)
  add_executable(pw_swept_bounds_tests tests/unit/swept_bounds_tests.cpp)
  target_link_libraries(pw_swept_bounds_tests PRIVATE pw_core)
  add_test(NAME swept_bounds_reference COMMAND pw_swept_bounds_tests)
  add_executable(pw_swept_sphere_tests tests/unit/swept_sphere_tests.cpp)
  target_link_libraries(pw_swept_sphere_tests PRIVATE pw_core)
  add_test(NAME swept_sphere_reference COMMAND pw_swept_sphere_tests)
  add_executable(pw_settling_tests tests/unit/settling_tests.cpp)
  target_link_libraries(pw_settling_tests PRIVATE pw_core)
  add_test(NAME settling_hysteresis COMMAND pw_settling_tests)
  add_executable(pw_position_friction_tests tests/unit/position_friction_tests.cpp)
  target_link_libraries(pw_position_friction_tests PRIVATE pw_core)
  add_test(NAME position_friction_reference COMMAND pw_position_friction_tests)
endif()
if(PW_BUILD_APP)
  if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "The initial GPU application supports macOS and Windows only. Use PW_BUILD_APP=OFF for CPU tests.")
  endif()
  include(cmake/Dependencies.cmake)
  add_library(pw_assets STATIC src/assets/Materials.cpp src/assets/Scenes.cpp src/assets/Recordings.cpp src/assets/RecordingSession.cpp)
  target_link_libraries(pw_assets PUBLIC pw_core PRIVATE nlohmann_json::nlohmann_json)
  add_executable(pw_asset_builder tools/asset_builder/main.cpp)
  target_link_libraries(pw_asset_builder PRIVATE pw_assets)
  if(BUILD_TESTING)
    add_executable(pw_recording_tests tests/unit/recording_tests.cpp)
    target_link_libraries(pw_recording_tests PRIVATE pw_assets nlohmann_json::nlohmann_json)
    add_test(NAME recording_validation COMMAND pw_recording_tests ${PROJECT_SOURCE_DIR}/assets ${CMAKE_BINARY_DIR}/recording-test)
    add_executable(pw_scene_tests tests/unit/scene_tests.cpp)
    target_link_libraries(pw_scene_tests PRIVATE pw_assets nlohmann_json::nlohmann_json)
    add_test(NAME scene_validation COMMAND pw_scene_tests ${PROJECT_SOURCE_DIR}/assets)

    target_link_libraries(pw_material_tests PRIVATE pw_assets nlohmann_json::nlohmann_json)
    add_test(NAME material_validation COMMAND pw_material_tests ${PROJECT_SOURCE_DIR}/assets/materials/default.json)
  endif()
  add_subdirectory(src/app)
endif()
