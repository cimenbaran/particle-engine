cmake_minimum_required(VERSION 3.28)
project(ParticleWorld VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS")
endif()
option(PW_BUILD_APP "Build the native GPU application" ON)
include(CTest)
add_library(pw_core INTERFACE)
target_include_directories(pw_core INTERFACE src)
target_compile_features(pw_core INTERFACE cxx_std_20)
if(BUILD_TESTING)
  add_executable(pw_core_tests tests/unit/core_tests.cpp)
  target_link_libraries(pw_core_tests PRIVATE pw_core)
  add_test(NAME core_invariants COMMAND pw_core_tests)
endif()
if(PW_BUILD_APP)
  if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "The initial GPU application supports macOS and Windows only. Use PW_BUILD_APP=OFF for CPU tests.")
  endif()
  include(cmake/Dependencies.cmake)
  add_subdirectory(src/app)
endif()
